cmake_minimum_required(VERSION 3.16)
project(mstar_cpu_encoder LANGUAGES C CXX)
if(NOT IS_DIRECTORY "${GGML_SOURCE}/include")
    message(FATAL_ERROR "GGML_SOURCE must name the pinned ggml source directory")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
add_subdirectory("${GGML_SOURCE}" ggml EXCLUDE_FROM_ALL)
add_library(mstar_cpu_encoder SHARED encoder.cpp decoder.cpp attention_probe.cpp)
target_compile_features(mstar_cpu_encoder PRIVATE cxx_std_17)
# Physical callbacks reuse the pinned compiled CPU ADD/GELU entry points.
target_include_directories(mstar_cpu_encoder PRIVATE "${GGML_SOURCE}/src")
target_compile_options(mstar_cpu_encoder PRIVATE -fno-fast-math)
target_link_libraries(mstar_cpu_encoder PRIVATE ggml ggml-cpu ggml-base)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # Prevent interposition with an independently loaded GPU ggml provider.
    target_link_options(mstar_cpu_encoder PRIVATE -Wl,--exclude-libs,ALL -Wl,--no-undefined)
endif()

option(MSTAR_ENCODER_QK_SDOT "Encoder-only transient Q8 QK with F32 tiled softmax/PV" OFF)
if(MSTAR_ENCODER_QK_SDOT)
    set_property(SOURCE encoder.cpp APPEND PROPERTY COMPILE_DEFINITIONS MSTAR_ENCODER_QK_SDOT=1)
endif()
set(MSTAR_ENCODER_Q8_CONV "0" CACHE STRING "Physical Q8 convolution mask: 0=F16, 1=conv1, 2=conv2, 3=both")
set_property(CACHE MSTAR_ENCODER_Q8_CONV PROPERTY STRINGS 0 1 2 3)
if(NOT MSTAR_ENCODER_Q8_CONV MATCHES "^[0-3]$")
    message(FATAL_ERROR "MSTAR_ENCODER_Q8_CONV must be 0, 1, 2 or 3")
endif()
if(NOT MSTAR_ENCODER_Q8_CONV STREQUAL "0")
    set_property(SOURCE encoder.cpp APPEND PROPERTY COMPILE_DEFINITIONS MSTAR_ENCODER_Q8_CONV=${MSTAR_ENCODER_Q8_CONV})
endif()

option(MSTAR_ENCODER_PV_SDOT "Encoder-only transient Q8 PV" OFF)
if(MSTAR_ENCODER_PV_SDOT)
    if(NOT MSTAR_ENCODER_QK_SDOT)
        message(FATAL_ERROR "PV SDOT requires encoder QK SDOT")
    endif()
    set_property(SOURCE encoder.cpp APPEND PROPERTY COMPILE_DEFINITIONS MSTAR_ENCODER_PV_SDOT=1)
endif()

set(MSTAR_ENCODER_FFN_TILE "0" CACHE STRING "Physical FFN query tile: 0=full query, 128=local complete blocks")
set_property(CACHE MSTAR_ENCODER_FFN_TILE PROPERTY STRINGS 0 128)
if(NOT MSTAR_ENCODER_FFN_TILE MATCHES "^(0|128)$")
    message(FATAL_ERROR "MSTAR_ENCODER_FFN_TILE must be 0 or 128")
endif()
if(NOT MSTAR_ENCODER_FFN_TILE STREQUAL "0")
    set_property(SOURCE encoder.cpp APPEND PROPERTY COMPILE_DEFINITIONS MSTAR_ENCODER_FFN_TILE=${MSTAR_ENCODER_FFN_TILE})
endif()
